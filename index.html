const MAX_TEXT_LENGTH = 5000; const TIMEOUT_MS = 12000; const safeJson = async (response) => { try { return await response.json(); } catch { return null; } }; const makeAbortable = () => { const controller = new AbortController(); const timer = setTimeout(() => controller.abort(), TIMEOUT_MS); return { signal: controller.signal, clear: () => clearTimeout(timer) }; }; async function googleTranslate(text, source, target) { const url = new URL('https://translate.googleapis.com/translate_a/single'); url.searchParams.set('client', 'gtx'); url.searchParams.set('sl', source === 'auto' ? 'auto' : source); url.searchParams.set('tl', target); url.searchParams.set('dt', 't'); url.searchParams.set('q', text); const request = makeAbortable(); try { const response = await fetch(url, { signal: request.signal, headers: { Accept: 'application/json' } }); if (!response.ok) throw new Error('Google unavailable'); const data = await safeJson(response); if (!Array.isArray(data?.[0])) throw new Error('No translation'); const translation = data[0] .filter((part) => Array.isArray(part) && typeof part[0] === 'string') .map((part) => part[0]) .join(''); if (!translation) throw new Error('Empty translation'); return { translation, detectedLanguage: data?.[2] || source, provider: 'Google Translate web engine' }; } finally { request.clear(); } } async function memoryTranslate(text, source, target) { const url = new URL('https://api.mymemory.translated.net/get'); url.searchParams.set('q', text); url.searchParams.set('langpair', `${source === 'auto' ? 'en' : source}|${target}`); url.searchParams.set('mt', '1'); const request = makeAbortable(); try { const response = await fetch(url, { signal: request.signal, headers: { Accept: 'application/json' } }); const data = await safeJson(response); const translation = data?.responseData?.translatedText; if (!translation || !response.ok || data?.responseStatus !== 200) { throw new Error('Fallback unavailable'); } return { translation, detectedLanguage: source, provider: 'MyMemory fallback' }; } finally { request.clear(); } } export default async function handler(req, res) { res.setHeader('Access-Control-Allow-Origin', '*'); res.setHeader('Access-Control-Allow-Headers', 'Content-Type'); res.setHeader('Access-Control-Allow-Methods', 'POST, OPTIONS'); if (req.method === 'OPTIONS') { return res.status(204).end(); } if (req.method !== 'POST') { return res.status(405).json({ error: 'Use POST /api/translate' }); } const { text, source = 'auto', target } = req.body || {}; if (!target || typeof text !== 'string' || !text.trim()) { return res.status(400).json({ error: 'text and target are required' }); } if (text.length > MAX_TEXT_LENGTH) { return res.status(413).json({ error: `Text is limited to ${MAX_TEXT_LENGTH} characters` }); } if (source === target) { return res.status(200).json({ translation: text, source, target, provider: 'same language' }); } try { const result = await googleTranslate(text, source, target).catch(() => memoryTranslate(text, source, target)); return res.status(200).json({ ...result, source, target }); } catch { return res.status(503).json({ error: 'Translation service is temporarily unavailable. Please try again.' }); } }
